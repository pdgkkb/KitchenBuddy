# Happy Bite — backend dependencies.
#
# Rebuilt from what the code actually imports, because the original was missing
# its core section: fastapi, pydantic-settings and friends were never listed,
# which is why a fresh venv started and then died on `No module named
# 'pydantic_settings'`.
#
#     python3.12 -m venv .venv          # 3.11 or 3.12 — NOT 3.13/3.14, see below
#     source .venv/bin/activate
#     pip install -r requirements.txt
#
# Everything below "the app itself" is optional. Leave a section out and that
# one feature switches off; the server still starts and /api/health/models says
# which package is absent.


# ---------------------------------------------------------------- the app itself
# Without these the server does not start at all.

fastapi>=0.110              # the web framework
uvicorn[standard]>=0.29     # the server. [standard] brings watchfiles, so --reload works
pydantic>=2.6               # request/response models, and the recipe schema
pydantic-settings>=2.2      # app/config.py — reads backend/.env
python-dotenv>=1.0          # what pydantic-settings uses to read that .env
httpx>=0.27                 # outbound HTTP: Ollama, link import, image downloads
python-multipart>=0.0.9     # REQUIRED for UploadFile — /speech/hear and /receipt/read
                            # are silent 422s without it
beautifulsoup4>=4.12        # app/importer.py — reading a recipe off a web page


# ---------------------------------------------------------------- chat providers
# You need ONE of these, matching LLM_PROVIDER in .env. `openai` also covers a
# local model served by Ollama or vLLM (LLM_PROVIDER=openai + OPENAI_BASE_URL),
# which is the fully-offline setup.

anthropic>=0.34             # LLM_PROVIDER=anthropic
openai>=1.40                # LLM_PROVIDER=openai, OpenAI images, OpenAI voice


# ---------------------------------------------------------------- local voice
# SPEECH_PROVIDER=local. Two models: one that hears you, one that speaks.
# Both download their own weights on first use.

faster-whisper>=1.0         # hearing you (CTranslate2 Whisper)
kokoro>=0.7                 # speaking to you — 82M, far better than Piper
soundfile>=0.12             # how Kokoro's audio gets out as WAV
misaki[en]>=0.7             # Kokoro's phonemiser
                            # also: brew install espeak-ng

# Lighter alternative to Kokoro, only if you set TTS_ENGINE=piper. Comment it
# out if you're not using it — it pulls onnxruntime.
piper-tts>=1.2


# ---------------------------------------------------------------- local pictures
# IMAGE_PROVIDER=local. Stable Diffusion Turbo through diffusers: free, no key,
# runs on the Mac's GPU. ~2.5 GB downloads on first use. This is the heavy part
# of the install — drop the block and set IMAGE_PROVIDER=none or openai.

torch>=2.2
diffusers==0.35.2           # compatible with the Transformers 4.x API
transformers==4.57.6
huggingface-hub==0.36.2
accelerate>=0.30
safetensors>=0.4
pillow>=10.0                # also used to straighten receipt photos before OCR


# ---------------------------------------------------------------- reading receipts
# The character recogniser. pytesseract is only the binding — the tesseract
# program itself comes from Homebrew:
#     brew install tesseract tesseract-lang

pytesseract>=0.3.10


# ---------------------------------------------------------------- tests
# cd backend && python -m pytest -q

pytest>=8.0
pytest-asyncio>=0.23


# ---------------------------------------------------------------- one warning
# torch, diffusers and kokoro publish NO wheels for Python 3.13 or 3.14. On
# those versions pip either fails or tries to build from source and dies, and
# the app reports pictures and voice as permanently off. Build the venv on 3.12.
